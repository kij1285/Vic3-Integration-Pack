    possible = {
        is_ai = no
        c:KDH ?= ROOT

        add_acceptance = { culture = cu:manchu   value = 10 }

        add_acceptance = { culture = cu:ainu     value = 10 }

        set_global_variable = vip_kdh_acceptance_done
    }

    ai_will_do = { always = no }
}
